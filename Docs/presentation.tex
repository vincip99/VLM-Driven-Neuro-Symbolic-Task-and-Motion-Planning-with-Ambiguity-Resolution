\begin{frame}[t]
\frametitle{Perception: Visual Grounding \& Ambiguity Resolution}
\footnotesize
Grounds natural language human instructions in the observed scene and explicitly reasons about task ambiguity through multi-turn visual dialogue. It eliminates hallucinated assumptions before symbolic deliberation

\vspace{0.1cm}
\begin{columns}[T]
    \begin{column}{0.55\textwidth}
        \begin{block}{Ambiguity Resolution Pipeline}
            \begin{itemize}\scriptsize
                \item \textbf{VLM Visual Scene Grounding (Qwen 2.5-VL):}
                Parses the camera frame into entity schemas:
                \begin{itemize}\tiny
                    \item \texttt{task\_objects}: Manipulable items (\textit{e.g., cans, cubes}).
                    \item \texttt{target\_locations}: Receptacles (\textit{e.g., sorting bin, pot}).
                \end{itemize}
                
                \item \textbf{Ambiguity Detection:}
                Evaluates query against detected scene entities:
                \begin{itemize}\tiny
                    \item Flags referential ambiguity (\textit{e.g., multiple cubes present}).
                    \item Schema: \texttt{AmbiguityReasoning(task\_ambiguous, question)}.
                \end{itemize}

                \item \textbf{Human-in-the-Loop Clarification:}
                Prompts the user with a targeted question, appends response to chat history, and disambiguates the scene.

                \item \textbf{Symbols Grounding:}
                Outputs clean symbols $\{\text{Objects}, \text{Locations}\} \subset O_{\text{obj}}, O_{\text{loc}}$ directly to the LLaMA-3 PDDL synthesizer.
            \end{itemize}

\begin{frame}[t]
\frametitle{VLM Architectures: Dual-Encoder Foundations}
\footnotesize
Vision-Language Models (VLMs) integrate visual perception with linguistic reasoning by bridging continuous visual feature spaces and discrete language token spaces.

\vspace{0.2cm}
\begin{columns}[T]
    \begin{column}{0.48\textwidth}
        \begin{figure}
            \centering
            % Adjusting width slightly for better margin breathing room
            \includegraphics[width=0.95\linewidth]{pictures/llama3.png}
            \vspace{-0.1cm}
            \caption{\scriptsize Standard Dual-Encoder VLM Architecture}
        \end{figure}
    \end{column}

    \begin{column}{0.5\textwidth}
        \setbeamercolor{vlmbox}{bg=darkblue,fg=white}
        
        \begin{beamercolorbox}[sep=5pt,shadow=true,rounded=true]{vlmbox}
            \textbf{Language Encoder (Transformer)}
            \vspace{0.1cm} 
            \scriptsize
            \begin{itemize}
                \setlength\itemsep{0.2em}
                \item Transforms discrete sequences into continuous numerical \textbf{embeddings}.
                \item Utilizes \textbf{self-attention mechanisms} to dynamically weigh token importance.
                \item \textbf{Decoders} auto-regressively generate the most statistically probable sequence.
            \end{itemize}
        \end{beamercolorbox}
        
        \vspace{0.3cm}
        
        \begin{beamercolorbox}[sep=5pt,shadow=true,rounded=true]{vlmbox}
            \textbf{Vision Encoder (CNN / ViT)}
            \vspace{0.1cm}
            \scriptsize
            \begin{itemize}
                \setlength\itemsep{0.2em}
                \item Extracts hierarchical spatial features via Convolutional Neural Networks or Vision Transformers.
                \item ViTs segment images into non-overlapping patches, processing them as sequential tokens.
            \end{itemize}
        \end{beamercolorbox}
    \end{column}
\end{columns}
\end{frame}


\begin{frame}[t]
    \frametitle{VLM Training Methodologies}
    \footnotesize
    Training fuses information from vision and language encoders, correlating visual features with text semantics to enable unified multimodal reasoning.

    \vspace{0.3cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Contrastive Learning}
                \scriptsize
                Aligns modalities in a shared embedding space. The model minimizes the latent distance between matching image-text pairs while maximizing it for non-matching pairs.
            \end{block}
            
            \vspace{0.1cm}
            \begin{block}{Masked Modeling}
                \scriptsize
                Forces the model to reconstruct randomly obscured image patches or masked text tokens, learning robust contextual representations.
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{Generative Training}
                \scriptsize
                Optimizes the model to auto-regressively generate text conditioned on visual inputs (e.g., image captioning or visual question answering).
            \end{block}

            \vspace{0.1cm}
            \begin{block}{Pretrained Projection}
                \scriptsize
                Freezes a pretrained LLM and Vision Encoder, training only a lightweight mapping network (e.g., MLP projector) to translate visual embeddings into the LLM's input space.
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[t]
\frametitle{Symbolic Planning: PDDL Formulation}
\footnotesize
\textbf{Symbolic Planning} uses formal logic to find a sequence of actions that transition a system from an initial state to a target goal state. \textbf{PDDL} (Planning Domain Definition Language) is the standard modeling language for this, decoupling the robot's physical capabilities from specific task specifications to guarantee causal correctness before execution.

\vspace{0.15cm}
\begin{columns}[T]
        \begin{column}{0.50\textwidth}
        \begin{block}{Formal Planning Tuple $\mathcal{P} = (\mathcal{D}, O, s_0, G)$}
            \begin{itemize}\scriptsize
                \item \textbf{Planning Domain $\mathcal{D} = (\mathcal{L}, \mathcal{A})$ [Static]:}
                \begin{itemize}\tiny
                    \item $\mathcal{L}$: Predicates describing state relations.
                    \item $\mathcal{A}$: Action schemas with preconditions and effects.
                    \item \textit{Fixed and verified offline} (robot capability bounds).
                \end{itemize}
                \vspace{0.06cm}
                \item \textbf{Planning Problem Instance [Dynamic]:}
                \begin{itemize}\tiny
                    \item $O$: Grounded universe of objects and locations.
                    \item $s_0$: Completely grounded initial state from VLM.
                    \item $G$: Target goal conditions from user task.
                    \item \textit{Synthesized online} via LLaMA~3 from perception.
                \end{itemize}
            \end{itemize}
        \end{block}
        
        \vspace{0.12cm}
        \setbeamercolor{verdictbox}{bg=darkblue,fg=white}
        \begin{beamercolorbox}[sep=4pt,center,shadow=true,rounded=true]{verdictbox}
            \tiny \textbf{Key Principle:} Domain physics are fixed; only state and goal are queried online.
        \end{beamercolorbox}
    \end{column}
    
    \begin{column}{0.48\textwidth}
        \begin{block}{Manipulation Domain Definition (\texttt{domain.pddl})}
            \setbeamercolor{termbox}{bg=white,fg=black}
            \begin{beamercolorbox}[sep=4pt,rounded=true,shadow=true,wd=\linewidth]{termbox}
                \ttfamily\tiny
                ; Grounded State Predicates $\mathcal{L}$\\
                (holding ?obj)\\
                (on-table ?obj)\\
                (on ?obj ?loc)\\[0.15cm]
                ; Robotic Action Schemas $\mathcal{A}$\\
                (:action pick\\
                \hspace{2mm} :parameters (?o - obj)\\
                \hspace{2mm} :precondition (and (on-table ?o)\\
                \hspace{14mm} (not (holding ?o)))\\
                \hspace{2mm} :effect (and (holding ?o)\\
                \hspace{10mm} (not (on-table ?o))))\\[0.15cm]
                (:action place\\
                \hspace{2mm} :parameters (?o - obj ?loc - location)\\
                \hspace{2mm} :precondition (holding ?o)\\
                \hspace{2mm} :effect (and (on ?o ?loc)\\
                \hspace{10mm} (not (holding ?o))))
            \end{beamercolorbox}
        \end{block}
    \end{column}
\end{columns}
\end{frame}

\begin{frame}[t]
\frametitle{Neuro-symbolic planning: PDDL Problem Synthesis}
\footnotesize
While domain physics $\mathcal{D}$ are fixed offline, the planning problem $\mathcal{P} = (\mathcal{D}, O, s_0, G)$ is formulated online. \textbf{LLaMA~3} translates visually grounded symbols and task intent into $(O, s_0, G)$ to complete $\mathcal{P}$.

\vspace{0.05cm}
\begin{columns}[T]
    \begin{column}{0.50\textwidth}
        \begin{block}{Problem Synthesis Workflow}
            \begin{itemize}\scriptsize
                \setlength\itemsep{0.45em}
                \item \textbf{1. Grounded Few-Shot Prompting:} Uses grounded entities ($O_{\text{obj}}, O_{\text{loc}}$), scene state $s_0$, and task $G$ alongside domain $\mathcal{D}$. Enforces strict typing ($O_{\text{obj}} \to \texttt{obj}$, $O_{\text{loc}} \to \texttt{location}$) and maps task goals to domain predicates.
                \item \textbf{2. Pydantic Validation, Auto-Repair \& Compilation:} Validates LLM output against pydantic \texttt{Problem} schema, auto-correct undeclared symbols, and retries on predicate violations. \texttt{Problem.to\_pddl()} deterministically serializes the verified instance into standard PDDL syntax.
            \end{itemize}
        \end{block}

        \vspace{0.05cm}
        \setbeamercolor{verdictbox}{bg=darkblue,fg=white}
        \begin{beamercolorbox}[sep=3pt,center,shadow=true,rounded=true]{verdictbox}
            \tiny \textbf{Key Guarantee:} Structured JSON + Pydantic validation eliminates LLM syntax hallucinations before reaching the planner.
        \end{beamercolorbox}
    \end{column}

    \begin{column}{0.48\textwidth}
        \begin{block}{Generated Planning Problem (\texttt{problem.pddl})}
            \setbeamercolor{termbox}{bg=white,fg=black}
            \begin{beamercolorbox}[sep=3pt,rounded=true,shadow=true,wd=\linewidth]{termbox}
                \ttfamily\tiny
                ; Online problem instance $\mathcal{P} = (\mathcal{D}, O, s_0, G)$\\
                (define (problem manipulation-task)\\
                \hspace{2mm} (:domain manipulation)\\
                \hspace{2mm} (:objects\\
                \hspace{4mm} red\_can yellow\_cube blue\_can - obj\\
                \hspace{4mm} sorting\_bin pot - location\\
                \hspace{2mm} )\\
                \hspace{2mm} (:init\\
                \hspace{4mm} (on-table red\_can)\\
                \hspace{4mm} (on-table yellow\_cube)\\
                \hspace{4mm} (on-table blue\_can)\\
                \hspace{2mm} )\\
                \hspace{2mm} (:goal (and\\
                \hspace{4mm} (on red\_can sorting\_bin)\\
                \hspace{4mm} (on yellow\_cube pot)\\
                \hspace{4mm} (on blue\_can sorting\_bin)\\
                \hspace{2mm} ))\\
                )
            \end{beamercolorbox}
        \end{block}
    \end{column}
\end{columns}
\end{frame}


\begin{frame}[t]
    \frametitle{Plan Verification: Fast Downward Solver}
    \footnotesize
    \textbf{Fast Downward} acts as the symbolic deliberation engine, using initial states $s_0$, final goal $G$ and Domain specification $\mathcal{D}$ to obtain a verified action sequences via \textbf{Heuristic Forward State-Space Search}.

    \vspace{0.05cm}
    \begin{columns}[T]
        \begin{column}{0.53\textwidth}
            \vspace{-0.05cm}
            \begin{center}
                \makebox[\linewidth][c]{\includegraphics[width=1.08\linewidth]{pictures/deliberation_module.pdf}}
            \end{center}
            
            \vspace{0.12cm}
            \setbeamercolor{verdictbox}{bg=darkblue,fg=white}
            \begin{beamercolorbox}[sep=4pt,center,shadow=true,rounded=true]{verdictbox}
                \scriptsize \textbf{Deliberation Guarantee:} Computes verified causal action plans in $\mathbf{<0.08\,\text{s}}$ with zero hallucination.
            \end{beamercolorbox}
        \end{column}

        \begin{column}{0.45\textwidth}
            \begin{block}{Forward Search \& Heuristics}
                \begin{itemize}\scriptsize
                    \setlength\itemsep{0.20em}
                    \item \textbf{Search Paradigm:} Progresses forward in state space $\mathcal{S}$: $s_0 \xrightarrow{a_1} s_1 \to \dots \to s_g \models G$.
                    \item \textbf{Direct Grounding:} Grounded $s_0$ from VLM makes forward state progression directly evaluable.
                    \item \textbf{State Transitions:} Applies applicable action $a$ ($\text{pre}(a) \subseteq s$):
                    $s' = (s \setminus \text{del}(a)) \cup \text{add}(a)$.
                    \vspace{0.06cm}
                    \item \textbf{Heuristic Engine (LAMA / $h_{\text{FF}}$):}
                    \begin{itemize}\scriptsize
                        \setlength\itemsep{0.15em}
                        \item \textbf{Lazy GBFS:} Greedy best-first search for high-speed plan discovery.
                        \item \textbf{Dual Heuristics ($h_{\text{FF}} + h_{\text{LM}}$):} Delete-relaxation ($\text{del}(a) = \emptyset$) + causal Landmarks.
                        \item \textbf{Preferred Operators:} Prunes search branching via helpful actions.
                    \end{itemize}
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[t]
    \frametitle{PDDL Problem Generation experiments}
    \footnotesize
    To bridge perceptual symbols and symbolic planning, the deliberation layer couples \textbf{Pydantic data models} with programmatic domain validation and a reflection loop to eliminate hallucinations.

    \vspace{0.04cm}
    \begin{columns}[T]
        \begin{column}{0.50\textwidth}
            \begin{block}{Structured Pipeline \& Pydantic Schemas}
                \begin{itemize}\scriptsize
                    \setlength\itemsep{0.10em}
                    \item \textbf{Synthesizer Class (\texttt{AmbresStructured}):}
                    \newline \texttt{generate\_problem\_json(...)} queries LLaMA~3 with $(s_0, G, O_{\text{obj}}, O_{\text{loc}}, \mathcal{D})$ under few-shot JSON constraints.
                    \item \textbf{Data Models (\texttt{src.llm2pddl.problem}):}
                    \begin{itemize}\scriptsize
                        \setlength\itemsep{0.02em}
                        \item \texttt{Problem}: Declares \texttt{objects}, \texttt{init}, and \texttt{goal}.
                        \item \texttt{PDDLObject}: Type enforcement (\texttt{obj}, \texttt{location}).
                        \item \texttt{Predicate}: Structured facts with arity validation.
                    \end{itemize}
                    \item \textbf{Compilation (\texttt{Problem.to\_pddl()}):}
                    \newline Deterministically convert the JSON payload schema into standard PDDL formatted text.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{Closed-Loop Validation \& Auto-Repair}
                \begin{itemize}\scriptsize
                    \setlength\itemsep{0.10em}
                    \item \textbf{Validation Function:}
                    \newline \texttt{generate\_problem\_with\_retry($\dots, \mathcal{D}$)} verifies predicates and types against $\mathcal{D}$ via regex extraction.
                    \item \textbf{Handled Failure Modes:}
                    \begin{itemize}\scriptsize
                        \setlength\itemsep{0.02em}
                        \item \textit{Predicate Hallucination:} e.g., \texttt{(in ?x ?bin)} $\notin \mathcal{D}$.
                        \item \textit{Undeclared Symbols:} Missing object in \texttt{objects}.
                        \item \textit{Type Inversion:} Receptacle labeled as \texttt{obj}.
                    \end{itemize}
                    \item \textbf{Auto-Healing \& Reflection:}
                    \begin{itemize}\scriptsize
                        \setlength\itemsep{0.02em}
                        \item Inferred auto-declaration for missing receptacles.
                        \item Re-prompts LLaMA~3 with exact error traces (max 5 retries).
                    \end{itemize}
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\frametitle{BT Builder: Goal Checking and Tree Generation}
\begin{columns}[T]
    \begin{column}{0.53\textwidth}
        \small \textbf{3. Goal-Guarded Wrapper (\texttt{wrap\_with\_goal\_check})}
        \scriptsize
        A plain sequence is open-loop. If an object drops, it fails blindly. We wrap it in a reactive fallback loop:
        \begin{itemize}
            \item \textbf{Pre-Check (\texttt{GoalCheck\_Pre})}: Verifies if goal is already met before moving. If yes $\to$ succeeds immediately (no redundant actions).
            \item \textbf{Selector (\texttt{memory=False})}: Re-evaluates continuously on every tick for dynamic preemption.
            \item \textbf{Post-Check (\texttt{GoalCheck\_Post})}: Confirms physical completion after the sequence finishes.
            \item \textbf{Retry Loop (\texttt{Retry})}: If the grasp slips and post-check fails, automatically retries the plan up to $N$ times.
        \end{itemize}

        \small \textbf{4. Main Entrypoint (\texttt{build\_bt\_from\_pddl\_plan})}
        \scriptsize
        Single builder API providing two execution modes:

\begin{frame}[fragile]
    \frametitle{Path Planning: RRT}
    Rapidly Random Trees (RRT) is a Probabilistic Path Planning method used to efficiently construct a path in continuous spaces.
    \begin{algorithm}[H]
        \captionsetup{font=scriptsize}
        \scriptsize
        \caption{Goal Oriented RRT Algorithm}
        \begin{algorithmic}[1]
            \State $G.\text{init}(x_{\text{start}})$
            \For{$i = 1 \dots K$}
                \State $p = \text{UniformRandom}(0, 1)$ 
                \If{$p \leq p_{\text{goal}}$}
                    \State $x_{\text{rand}} \gets x_{\text{goal}}$
                \Else
                    \State $x_{\text{rand}} \gets \text{RandPose}()$
                \EndIf
                \State $x_{\text{nearest}} \gets \text{NearestVertex}(x_{\text{rand}}, G)$
                \State $x_{\text{new}} \gets \text{NewConf}(x_{\text{nearest}}, x_{\text{rand}}, \Delta x)$
                \If{\text{CollisionFree}$(x_{\text{nearest}}, x_{\text{new}})$}
                    \State $G.\text{add\_vertex}(x_{\text{new}})$
                    \State $G.\text{add\_edge}(x_{\text{nearest}}, x_{\text{new}})$
                    \If{$x_{\text{new}} \in X_{\text{goal}}$}
                        \Return \text{Path Found}
                    \EndIf
                \EndIf
            \EndFor
            \State \Return $G$
        \end{algorithmic}
    \end{algorithm}
    The goal oriented version samples the goal point with a given probability $p$ while a random point with probability $1-p$. 
\end{frame}

\begin{frame}
    \frametitle{Trajectory Tracking}
    \begin{itemize}
        \item \textbf{Cartesian Setpoints:} 
        The waypoints generated by the task-space RRT are Cartesian poses used as setpoints for a \textbf{Proportional (P) Controller}.
        \item \textbf{Tracking Error:} 
        At each time step, the position error is defined as:
        \[
            \mathbf{e} = \mathbf{p}_{\text{target}} - \mathbf{p}_{\text{EEF}}
        \]
        \item \textbf{Control Law:} 
        The controller outputs velocity commands bounded by a maximum speed:
        \[
            \mathbf{v}_{\text{cmd}} = \operatorname{clip}(K_p \mathbf{e},\, -v_{\max},\, v_{\max})
        \]
        which are sent to the underlying operational space controller.
        \item \textbf{Application:} 
        Used specifically during the \textbf{approach phase} in the pick action and during the \textbf{place action}. The place skill itself is completely based on the RRT planner and the trajectory tracker, while the pick skill is composed by more modules and need further explaination.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Proximal Policy Optimization}
    Proximal Policy Optimization is an on-policy actor-critic Reinforcement Learning method. PPO updates policy parameters $\theta_{k+1}$ by maximizing the \textbf{Clipped Surrogate Objective}:
    \[
    L^{\text{CLIP}}(\theta) = \hat{\mathbb{E}}_t \left[ \min \left( r_t(\theta)\hat{A}_t, \; \operatorname{clip}(r_t(\theta), 1-\epsilon, 1+\epsilon)\hat{A}_t \right) \right]
    \]
    
    where the probability ratio is defined as:
    \[
    r_t(\theta) = \frac{\pi_\theta(a_t \mid s_t)}{\pi_{\theta_k}(a_t \mid s_t)}
    \]

    \begin{itemize}
        \item \textbf{Clipped Objective:} The hyperparameter $\epsilon \approx 0.2$ penalizes moving $r_t(\theta)$ outside $[1-\epsilon, 1+\epsilon]$, preventing destructively large policy updates.
        \item \textbf{Actor-Critic Architecture:} Jointly optimizes a policy network $\pi_\theta(a|s)$ and a state-value baseline $V_\phi(s)$ using Generalized Advantage Estimation (GAE).
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Manipulation: PPO Approach}
    \small
    \texttt{TaskSpaceRRT} provides coarse global navigation to a pre-grasp hover pose ($\sim 18\,\text{cm}$ above the object). The \textbf{PPO policy} then takes over using a minimal, translation-invariant \textbf{11D observation vector}:
    
    \vspace{0.15cm}
    \centering
    $\mathbf{o}_t = \Big[\; \underbrace{\Delta\mathbf{p}}_{\mathbb{R}^3},\;\; \underbrace{\mathbf{q}_{\text{grip}}}_{\mathbb{R}^2},\;\; \underbrace{\dot{\mathbf{q}}_{\text{grip}}}_{\mathbb{R}^2},\;\; \underbrace{z_{\text{rel}}^{\text{EEF}}}_{\mathbb{R}^1},\;\; \underbrace{\Delta z_{\text{obj}}}_{\mathbb{R}^1},\;\; \underbrace{c_{\text{grasp}}}_{\{0,1\}},\;\; \underbrace{m_{\text{geom}}}_{\{0,1\}} \;\Big]^\top \in \mathbb{R}^{11}$
    \vspace{0.2cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Kinematics \& Spatial Relations}
                \begin{itemize}\footnotesize
                    \item $\Delta\mathbf{p} = \mathbf{p}_{\text{obj}} - \mathbf{p}_{\text{EEF}}$: 3D relative vector (translation invariance across table).
                    \item $z_{\text{rel}}^{\text{EEF}} = p_{\text{EEF}, z} - z_{\text{table}}$: Table clearance to avoid surface collisions.
                    \item $\Delta z_{\text{obj}} = p_{\text{obj}, z} - p_{\text{obj}, z}^{(0)}$: Vertical displacement measuring lift progress.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Gripper and Auxiliary States}
                \begin{itemize}\footnotesize
                    \item $\mathbf{q}_{\text{grip}}, \dot{\mathbf{q}}_{\text{grip}}$: Left/right finger joint positions and velocities (contact stall detection).
                    \item $c_{\text{grasp}} \in \{0, 1\}$: Bilateral force-closure flag (Robosuite contact normal check).
                    \item $m_{\text{geom}} \in \{0, 1\}$: Shape token ($0$: cube, $1$: can) enabling multi-geometry conditioning.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Manipulation: Reward Function}
    \small
    The agent is trained with a multi-stage continuous-discrete reward guiding the transition from approach $\to$ grasp $\to$ lift:
    \begin{equation*}
        R_t = r_{\text{reach}} + r_{\text{grip}} + r_{\text{grasp}} + r_{\text{lift}} - c_{\text{step}} + r_{\text{terminal}}
    \end{equation*}
    \vspace{-0.2cm}
    \begin{columns}[T]
        \begin{column}{0.50\textwidth}
            \begin{block}{Continuous Shaping Terms}
                \begin{itemize}\scriptsize
                    \item \textbf{Smooth Reaching ($d = \|\Delta\mathbf{p}\|_2$):}
                    \newline $r_{\text{reach}} = 1.0 - \tanh(10.0 \cdot d)$
                    \item \textbf{Grasp Maintenance:}
                    \newline $r_{\text{grasp}} = 2.0 \cdot \mathbb{I}(\text{grasped})$
                    \item \textbf{Conditioned Vertical Lift:}
                    \newline $r_{\text{lift}} = 20.0 \cdot \max(0, \Delta z_{\text{obj}}) \cdot \mathbb{I}(\text{grasped})$
                    \newline \textit{(Avoids lifting with an empty gripper)}
                    \item \textbf{Alive Cost:} $c_{\text{step}} = 0.01$ (incentivizes speed)
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Action Shaping \& Terminals}
                \begin{itemize}\scriptsize
                    \item \textbf{Gripper Timing Incentive $r_{\text{grip}}$:}
                    $$r_{\text{grip}} = \begin{cases}
                        +0.5 & \text{if } d < 4\,\text{cm} \land a_g > 0 \\
                        -0.25 & \text{if } d < 4\,\text{cm} \land a_g \le 0 \\
                        -0.2 & \text{if } d \ge 4\,\text{cm} \land a_g > 0.2 \\
                        0 & \text{otherwise}
                    \end{cases}$$
                    \textit{Prevents premature closure before alignment.}
                    \item \textbf{Terminal Sparse Rewards:}
                    \newline $+15.0$ if $\Delta z_{\text{obj}} \ge 6\,\text{cm} \land \text{grasped}$ (\textbf{Success})
                    \newline $-2.0$ if $p_{\text{obj}, z} < z_{\text{table}} - 4\,\text{cm}$ (\textbf{Drop})
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

                \begin{itemize}\scriptsize
                    \item \textbf{Dynamic Impedance:} PPO modulates continuous velocities based on real-time force closure rather than rigid fixed gains ($K_p = 10$).
                    \item \textbf{Privileged vs Reactive:} Heuristic requires exact object coordinates; PPO relies only on local $\Delta\mathbf{p}$ and joint kinematics.
                    \item \textbf{Safety Guard:} Hybrid pick node falls back to heuristic if step budget ($\le 90$) expires.
                \end{itemize}
            \end{block}
        \end{column}

                \begin{itemize}\scriptsize
                    \item \textbf{Neuro-Symbolic Deliberation:} Autonomous extraction of $(s_0, G)$ from RGB frames and language, fortified by interactive disambiguation and auto-repair ($96.4\%$ solver feasibility).
                    \item \textbf{Formal Fast Downward Gatekeeper:} Guarantees causal plan validity in $<0.08\,\text{s}$ before physical dispatch.
